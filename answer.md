```python
         print(elem)

    return json.dumps({"list": subprocess.check_output("ls -l", shell=True, text=True)})
```

# Описание проекта Gigachat v2.5

Проект Gigachat представляет собой интерактивную систему, которая позволяет пользователям взаимодействовать с искусственным интеллектом через текстовый интерфейс. Основная цель системы – помогать пользователям выполнять различные задачи, предоставляя им возможность запрашивать информацию, выполнять расчеты и управлять файлами.

## Основные особенности:

- **Интерактивность**: Пользователь вводит запросы, система обрабатывает их и возвращает результат.
- **Поддержка функций**: Система поддерживает выполнение различных встроенных функций, таких как получение текущего времени, создание файлов, чтение и запись содержимого файлов.
- **Расширяемость**: Возможность добавления новых функций для расширения функциональности системы.
- **Отчетность**: Записи всех запросов и ответов сохраняются в файле `answer.md`.

## Структура проекта:

### Файлы:
- `gigachat_v2.5.py`: Основной скрипт, управляющий работой системы.
- `answer.md`: Файл, в который сохраняется история взаимодействия пользователя с системой.
- `readme.md`: Текущий документ с описанием проекта.
- Другие вспомогательные скрипты и модули.

### Функциональность:

1. **Получение текущего времени**:
   - Позволяет пользователю получать актуальное время.
   
2. **Создание файлов**:
   - Поддерживается создание новых файлов с заданным названием.
   
3. **Чтение и запись файлов**:
   - Позволяет читать содержимое существующих файлов и записывать новый текст в существующие или создаваемые файлы.
   
4. **Выполнение расчетов**:
   - Поддерживаются базовые математические операции, например, сложение чисел.

## Пример использования:

```python
# Запуск основной программы
python gigachat_v2.5.py
```

После запуска программы вы можете вводить запросы, на которые система будет реагировать соответствующим образом. Например:

- Для получения текущего времени введите команду: `"get current time"`
- Чтобы создать файл, используйте команду: `"create file my_new_file.txt"`
- Для чтения файла напишите: `"read file my_new_file.txt"` 

Система сохраняет историю ваших взаимодействий в файле `answer.md`.

## Заключение

Gigachat является гибким инструментом, который можно адаптировать под конкретные нужды пользователя путем добавления новых функций и расширений. Проект активно развивается и поддерживается сообществом разработчиков.

```python
# gigachat_v2.6.py

def evaluate_math_expression(expression):
    try:
        # Используем встроенную функцию eval() для оценки выражения
        result = eval(expression)
        return f"Результат вашего математического выражения: {result}"
    except Exception as e:
        return f"Произошла ошибка при оценке выражения: {e}"

if __name__ == "__main__":
    while True:
        user_input = input("Введите ваше математическое выражение (или 'exit' для выхода): ")
        
        if user_input.lower() == "exit":
            break
            
        print(evaluate_math_expression(user_input))
```

2. Если выражение корректное, возвращается результат.
3. Если возникает какая-либо ошибка, выводится соответствующее сообщение об ошибке.
4. Цикл позволяет пользователю вводить несколько выражений перед завершением программы.

```python
def read_file(name_file):
    try:
        with open(name_file, "r", encoding="utf-8") as file:
            content = file.read()
        return json.dumps({"res": content})
    except FileNotFoundError:
        return json.dumps({"error": f"Файл {name_file} не найден."})


def write_file(name_file, text):
    try:
        with open(name_file, "w", encoding="utf-8") as file:
            file.write(text)
        return json.dumps({"res": f"Файл {name_file} успешно обновлен."})
    except Exception as e:
        return json.dumps({"error": str(e)})

    return json.dumps({"time": datetime.now().strftime("%H:%M:%S")})
```

```python
def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n-i-1):
            if arr[j] > arr[j+1]:
                arr[j], arr[j+1] = arr[j+1], arr[j]
    return arr

# Пример использования
```

```python
def quicksort(arr):
    if len(arr) <= 1:
        return arr
    else:
        pivot = arr.pop()
        less = []
        greater = []
        for x in arr:
            if x <= pivot:
                less.append(x)
            else:
                greater.append(x)
        return quicksort(less) + [pivot] + quicksort(greater)

# Пример использования
my_list = [64, 34, 25, 12, 22, 11, 90]
sorted_list = quicksort(my_list.copy())
print("Отсортированный массив:", sorted_list)
```
